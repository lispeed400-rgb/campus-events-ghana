Universal Payment Gateway as a Service

Accept Mobile Money on Any Website

Integrate Ghana Mobile Money (MTN MoMo, Telecel Cash, AT Money) and Cards into your campus startup, e-commerce store, or ticketing portal. We handle Paystack processing and redirect customers back to your website with instant payment confirmation.

⚡ How It Works

1

Initialize Session

Your website makes a POST request to /.netlify/functions/gateway-init with the amount, customer email, and your callback_url.

2

Customer Pays

Redirect your customer to the returned checkout_url. Our hosted gateway loads Paystack for instant MoMo prompt or Card entry.

3

Instant Return

After successful payment, the customer is returned to your callback_url?status=success&transaction_id=GW-... so your website can fulfill the order.

💻 Code Examples

// 1. Initialize Gateway Payment
async function payWithCampusGateway() {
    const res = await fetch('https://campuseventghana.site/.netlify/functions/gateway-init', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            amount: 50.00, // GHS
            email: 'customer@gmail.com',
            customer_name: 'Kofi Mensah',
            merchant_name: 'Campus Tech Store',
            merchant_ref: 'ORDER-1029',
            title: 'Wireless Earbuds Purchase',
            callback_url: 'https://mystore.com/checkout/success',
            cancel_url: 'https://mystore.com/cart'
        })
    });

    const data = await res.json();
    if (data.success) {
        // Redirect customer to hosted checkout
        window.location.href = data.checkout_url;
    } else {
        alert('Payment initiation failed: ' + data.error);
    }
}
const axios = require('axios');

async function createCheckoutSession(req, res) {
    try {
        const response = await axios.post('https://campuseventghana.site/.netlify/functions/gateway-init', {
            amount: 75.00,
            email: req.body.email,
            customer_name: req.body.name,
            merchant_name: 'Legon Hall Week Committee',
            merchant_ref: 'TICKET-9921',
            title: 'Hall Week Mega Bash Ticket',
            callback_url: 'https://legonhall.com/tickets/verify',
            metadata: { student_id: '10982341' }
        });

        res.json({ checkout_url: response.data.checkout_url });
    } catch (err) {
        res.status(500).json({ error: err.response?.data?.error || err.message });
    }
}
